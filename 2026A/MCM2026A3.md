### Best Practices for Documenting Model Diffs
1.  **Focus on "Why", not just "What"**: A code diff shows *what* changed (lines of code). A model diff must explain the *mathematical reasoning* (e.g., "Linearity was too strong an assumption").
2.  **Quantify Impact**: Always mention if the change improved accuracy (RMSE, $R^2$) or robustness.
3.  **Group by Module**: Don't mix data cleaning changes with core equation changes.
4.  **Keep it Non-Code**: Even if you implemented it in Python, describe the change in terms of variables ($x, y, \beta$) and logic, not variable names (`df_train`, `numpy.array`).

  def parse_time_to_seconds(time_str: str) -> float:
    """解析HH:MM:SS.sss格式为秒"""

  def calc_metrics(y_true: np.ndarray, y_pred: np.ndarray) -> Dict[str, float]:
    """计算R2, RMSE, MAE"""

    if mask.sum() == 0:
        return {"R2": np.nan, "RMSE": np.nan, "MAE": np.nan}
    y_true = y_true[mask]

    ss_res = np.sum((y_true - y_pred) ** 2)
    ss_tot = np.sum((y_true - np.mean(y_true)) ** 2)
    r2 = 1 - ss_res / ss_tot if ss_tot > 0 else np.nan

  def ocv_model_pred(soc: float, k0: float, k1: float, k2: float, k3: float) -> float:
    """OCV-SOC模型（用于预测）。

    Args:
        soc: SOC标量。
        k0: 常数项。
        k1: 线性项系数。
        k2: ln(SOC)项系数。
        k3: ln(1-SOC)项系数。

    Returns:
        OCV电压。
    """
    soc = np.clip(soc, 0.01, 0.99)
    return k0 + k1 * soc + k2 * np.log(soc) + k3 * np.log(1 - soc)

  def predict_empty_time(soc_0: float, p_load_mw: float, params: Dict) -> Tuple[float, List, List]:
    """
    预测电池耗尽时间（带效率修正）
    
    参数:
        soc_0: 初始SOC
        p_load_mw: 负载功率 (mW)
        params: 电池参数
    
    返回:
        t_empty: 耗尽时间 (秒)
        soc_trajectory: SOC轨迹
        time_trajectory: 时间轨迹
    
    效率修正说明：
        实际电池放电过程中存在多种能量损失：
        1. 库仑效率损失 (~0.5%): 充放电过程中的不可逆电化学反应
        2. 内阻热损失 (~3-8%): I²R发热，特别是低SOC时内阻增大
        3. Peukert效应 (~1-2%): 大电流放电时有效容量降低
        4. 极化损失 (~1-2%): 浓差极化和电化学极化
        
        综合效率因子 η_energy ≈ 0.90-0.95
    """
    q_nom = params.get("Q_nom", 3.0)  # Ah
    soh = params.get("SOH", 1.0)
    r0 = params.get("R0", 0.024)  # Ω
    soc_min = params.get("SOC_min", 0.05)
    v_min = params.get("V_min", 2.8)
    dt = params.get("dt", 1.0)  # 秒
    
    k0 = params.get("K0", 3.5418)
    k1 = params.get("K1", 0.5600)
    k2 = params.get("K2", 0.1123)
    k3 = params.get("K3", -0.0144)
    
    # 新增：能量效率因子（考虑库仑效率、内阻热损失、Peukert效应）
    # 基础库仑效率
    eta_coulomb = params.get("eta_coulomb", 0.995)
    
    q_eff = q_nom * soh * 3600  # As
    p_load = p_load_mw / 1000  # W
    
    soc = soc_0
    t = 0
    t_max = 200 * 3600  # 最大200小时
    
    soc_traj = [soc_0]
    time_traj = [0]
    
    while soc > soc_min and t < t_max:
        v_oc = ocv_model_pred(soc, k0, k1, k2, k3)
        
        # SOC依赖的内阻增加（低SOC时内阻增大）
        # R(SOC) = R0 * (1.2 - 0.4*SOC) 简化模型
        r0_soc = r0 * (1.2 - 0.4 * soc)
        
        # 二次方程求解电流: P = V*I = (V_oc - I*R)*I
        # I² * R - I * V_oc + P = 0
        discriminant = v_oc**2 - 4 * r0_soc * p_load
        if discriminant < 0:
            break
        
        i = (v_oc - np.sqrt(discriminant)) / (2 * r0_soc)
        v = v_oc - i * r0_soc
        
        if v < v_min:
            break
        
        # 计算内阻热损失功率
        p_loss = i**2 * r0_soc  # W
        
        # 动态效率：考虑内阻损失占总功耗的比例
        # 实际消耗的能量 = 负载功率 + 内阻损耗
        # 效率 = P_load / (P_load + P_loss)
        eta_dynamic = p_load / (p_load + p_loss) if (p_load + p_loss) > 0 else 1.0
        
        # 更新SOC（考虑效率损失，实际消耗的库仑量更大）
        d_soc = -i * dt / q_eff / eta_coulomb
        soc = soc + d_soc
        t = t + dt
        
        # 每分钟记录一次
        if int(t) % 60 == 0:
            soc_traj.append(soc)
            time_traj.append(t)
    
    return t, soc_traj, time_traj
